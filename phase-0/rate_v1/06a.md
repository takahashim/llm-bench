```ruby
# app/models/user.rb
class User < ApplicationRecord
  validates :email, uniqueness: true
end

# app/services/create_user_service.rb
class CreateUserService
  def initialize(user_repository)
    @user_repository = user_repository
  end

  def call(email:, password:)
    user = @user_repository.create(email: email, password: password)
    Result::Success.new(user)
  rescue ActiveRecord::RecordInvalid => e
    Result::Failure.new(e.message)
  end
end

# app/services/user_service.rb
class UserService
  def initialize(user_repository)
    @user_repository = user_repository
  end

  def find_by_id(id:)
    @user_repository.find_by(id: id)
  end

  def update(id:, email:)
    user = @user_repository.find_by(id: id)
    return Result::Failure.new("User not found") if user.nil?

    user.email = email
    user.save!
    Result::Success.new(user)
  rescue ActiveRecord::RecordNotFound
    Result::Failure.new("User not found")
  end
end

# app/models/result.rb
class Result
  class Failure < self; end
  class Success < self; end
end

# app/repositories/user_repository.rb
class UserRepository
  def initialize(user_model)
    @user_model = user_model
  end

  def create(email:, password:)
    @user_model.create(email: email, password: password)
  end

  def find_by(id:)
    @user_model.find_by(id: id)
  end
end

# app/controllers/users_controller.rb
class UsersController < ApplicationController
  def create
    service = CreateUserService.new(UserRepository.new(User))
    begin
      result = service.call(email: params[:email], password: params[:password])
      if result.is_a?(Result::Success)
        @user = result.value
        render :show, status: :created
      else
        render json: { errors: result.message }, status: :unprocessable_entity
      end
    rescue => e
      logger.error e
      render json: { errors: e.message }, status: :internal_server_error
    end
  end

  def show
    service = UserService.new(UserRepository.new(User))
    @user = service.find_by_id(id: params[:id])
    render :show
  end

  def update
    service = UserService.new(UserRepository.new(User))
    begin
      result = service.update(id: params[:id], email: params[:email])
      if result.is_a?(Result::Success)
        @user = result.value
        render :show, status: :ok
      else
        render json: { errors: result.message }, status: :unprocessable_entity
      end
    rescue => e
      logger.error e
      render json: { errors: e.message }, status: :internal_server_error
    end
  end
end

# app/views/users/show.html.erb
<h1>User</h1>
<p><%= @user.email %></p>
```
